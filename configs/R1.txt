! show access-lists DMZ-IN numbers the entries 10-90 in this order
ip access-list extended DMZ-IN
 remark Replies to sessions opened from inside (DVWA -> users)
 permit tcp 192.168.99.0 0.0.0.255 192.168.0.0 0.0.255.255 established
 remark Ping replies for troubleshooting
 permit icmp 192.168.99.0 0.0.0.255 192.168.0.0 0.0.255.255 echo-reply
 remark Wazuh agent on Ubuntu-Web -> Wazuh manager (enrollment + logs)
 permit tcp host 192.168.99.10 host 192.168.30.10 eq 1514
 permit tcp host 192.168.99.10 host 192.168.30.10 eq 1515
 remark NTP from the DMZ to its own gateway (R1)
 permit udp 192.168.99.0 0.0.0.255 host 192.168.99.254 eq ntp
 remark DMZ may not start connections to internal networks
 deny   ip 192.168.99.0 0.0.0.255 192.168.0.0 0.0.255.255 log
 remark DMZ may not reach the firewall/transit network
 deny   ip 192.168.99.0 0.0.0.255 10.10.10.252 0.0.0.3 log
 remark Internet-bound traffic goes on to pfSense (blocked + logged there)
 permit ip 192.168.99.0 0.0.0.255 any
 remark Anything else = spoofed source
 deny   ip any any log
!
! Syslog to Wazuh (Phase 6). Level 6 (informational) includes ACL log messages (%SEC-6-...).
logging source-interface Ethernet0/1.30
logging host 192.168.30.10
logging trap informational
!
! NTP (Phase 6): R1 is the lab time source. There is no upstream (isolated lab).
ntp master 3

line con 0
